**Goal:** Entregar `forge qa` como 13º subcomando (revisita Decisão 9), gate adversarial multi-agente com 4 attack vectors (spec-vs-spec / coverage / chaos / validator-claim), 4 scope targets (feature / screen / task / paranoid), sandbox isolado pra validator-claim audit, e integração ortogonal com `forge evolve` via proposed-evolutions (Decisão 26).

# validators/validate_qa_report.py
"""Validator: qa-report.json schema (v1) — gerado por forge qa Phase 4.

Schema fonte: docs/schemas/qa-report.md.
"""

from __future__ import annotations

from typing import Any


class QAReportValidationError(ValueError):
    """qa-report.json não cumpre o schema v1."""


_VALID_VERDICTS = {"BLOCK", "FLAG", "PASS"}
_VALID_SCOPE_TYPES = {"feature", "screen", "task", "paranoid"}
_REQUIRED_SEVERITY_KEYS = {"critical", "high", "medium", "low", "info"}
_REQUIRED_VECTOR_KEYS = {"spec-vs-spec", "coverage", "chaos", "validator-claim"}


def validate_qa_report(data: dict[str, Any]) -> None:
    """Valida o dict carregado do qa-report.json. Raise em qualquer desvio.

    Política mentor-calma: mensagem de erro nomeia o campo + o que esperava.
    """
    if not isinstance(data, dict):
        raise QAReportValidationError("qa-report deve ser dict no topo")

    sv = data.get("schema_version")
    if sv != 1:
        raise QAReportValidationError(
            f"schema_version esperado=1, recebido={sv!r}. "
            f"Bump exige mudança breaking documentada."
        )

    for required in ("run", "verdict", "summary", "findings"):
        if required not in data:
            raise QAReportValidationError(f"campo obrigatório ausente: {required!r}")

    verdict = data["verdict"]
    if verdict not in _VALID_VERDICTS:
        raise QAReportValidationError(
            f"verdict={verdict!r} inválido. Valores aceitos: {sorted(_VALID_VERDICTS)}"
        )

    run = data["run"]
    if not isinstance(run, dict):
        raise QAReportValidationError("run deve ser dict")
    for required in ("id", "scope", "started_at", "finished_at", "duration_s"):
        if required not in run:
            raise QAReportValidationError(f"run.{required} ausente")

    scope = run["scope"]
    if scope.get("type") not in _VALID_SCOPE_TYPES:
        raise QAReportValidationError(
            f"run.scope.type={scope.get('type')!r} inválido. "
            f"Aceitos: {sorted(_VALID_SCOPE_TYPES)}"
        )

    summary = data["summary"]
    by_sev = summary.get("by_severity", {})
    if set(by_sev.keys()) != _REQUIRED_SEVERITY_KEYS:
        raise QAReportValidationError(

"""Validator: shape JSON individual de um qa-finding.

Schema fonte: docs/schemas/qa-finding.md.
"""

from __future__ import annotations

import re
from typing import Any


class QAFindingValidationError(ValueError):
    """qa-finding individual não cumpre o schema v1."""


_VALID_SEVERITY = {"critical", "high", "medium", "low", "info"}
_CORE_VECTORS = {"spec-vs-spec", "coverage", "chaos", "validator-claim"}
_FINGERPRINT_RE = re.compile(r"^[0-9a-f]{64}$")
_REQUIRED_TOP = {"id", "fingerprint", "vector", "severity", "title",
                 "description", "scope", "evidence", "proposed_evolution",
                 "created_at"}


def validate_qa_finding(data: dict[str, Any], known_extension_vectors: set[str] | None = None) -> None:
    """Valida shape individual. `known_extension_vectors` permite cards
    via qa-extensions registrarem vector names extras."""
    if not isinstance(data, dict):
        raise QAFindingValidationError("finding deve ser dict")

    missing = _REQUIRED_TOP - set(data.keys())
    if missing:
        raise QAFindingValidationError(
            f"finding com campos obrigatórios ausentes: {sorted(missing)}"
        )

    if not _FINGERPRINT_RE.match(data["fingerprint"]):
        raise QAFindingValidationError(
            f"fingerprint inválido (esperado sha256 lowercase): {data['fingerprint']!r}"
        )

    if data["severity"] not in _VALID_SEVERITY:
        raise QAFindingValidationError(
            f"severity={data['severity']!r} inválida; aceitos: {sorted(_VALID_SEVERITY)}"
        )

    vector = data["vector"]
    valid_vectors = _CORE_VECTORS | (known_extension_vectors or set())
    if vector not in valid_vectors:
        raise QAFindingValidationError(
            f"vector={vector!r} inválido. Core: {sorted(_CORE_VECTORS)}. "
            f"Extension registrados: {sorted(known_extension_vectors or [])}"
        )

    evidence = data["evidence"]
    if not isinstance(evidence, dict):
        raise QAFindingValidationError("evidence deve ser dict")

    if vector == "validator-claim" and "sandbox_result" not in evidence:
        raise QAFindingValidationError(
            "vector=validator-claim exige evidence.sandbox_result (subprocess executado)"
        )

    pe = data["proposed_evolution"]
    if not isinstance(pe, dict) or "type" not in pe:
        raise QAFindingValidationError("proposed_evolution.type ausente")
    if not pe["type"].startswith("qa-finding-"):
        raise QAFindingValidationError(
            f"proposed_evolution.type deve começar com 'qa-finding-'; recebido {pe['type']!r}"
        )

"""Validator: campo qa-extensions: em card.yaml — overlay-aware.

Schema fonte: docs/schemas/qa-extensions.md.
Política de colisão: canon×local → hard fail (Decisão 28 / Gap 5 Approach A).
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

from validators._common import load_catalog


class QAExtensionsValidationError(ValueError):
    """qa-extensions: bloco inválido em card.yaml."""


class QAExtensionsCollisionError(QAExtensionsValidationError):
    """Mesmo auditor name declarado em canon AND local — hard fail."""


_VALID_PHASES = {"static", "generative"}


def validate_qa_extensions(
    card_path: Path,
    card_data: dict[str, Any],
    *,
    other_cards: dict[str, dict[str, Any]] | None = None,
    project_root: Path | None = None,
) -> None:
    """Valida campo qa-extensions: do card. other_cards é o mapping
    {card_dir: card_data} dos outros cards já carregados (pra detectar
    colisão de auditor name).
    """
    qa_ext = card_data.get("qa-extensions")
    if qa_ext is None:
        return  # campo opcional

    auditors = qa_ext.get("auditors", [])
    if not isinstance(auditors, list):
        raise QAExtensionsValidationError(
            f"{card_path}: qa-extensions.auditors deve ser lista"
        )

    catalog = load_catalog(project_root) if project_root else {"labels": set()}
    catalog_labels = catalog.get("labels", set())

    for aud in auditors:
        name = aud.get("name")
        if not name:
            raise QAExtensionsValidationError(
                f"{card_path}: auditor sem campo `name`"
            )

        phase = aud.get("phase")
        if phase not in _VALID_PHASES:
            raise QAExtensionsValidationError(
                f"{card_path}: auditor {name!r} phase={phase!r} inválida; "
                f"aceitos: {sorted(_VALID_PHASES)} (Phase 0/3/4/5 são core-only)"
            )

        # Agents existem?
        agents = aud.get("contributes", {}).get("agents", [])
        for agent_rel in agents:
            agent_path = card_path.parent / agent_rel
            if not agent_path.exists():
                raise QAExtensionsValidationError(
                    f"{card_path}: auditor {name!r} referencia agent inexistente: {agent_path}"
                )

        # Requires capabilities no catálogo?
        for req in aud.get("requires", []):
            if req not in catalog_labels:
                raise QAExtensionsValidationError(
                    f"{card_path}: auditor {name!r} requires capability {req!r} "
                    f"não declarada no catálogo (canon ∪ local)"
                )

        # Colisão cross-card
        if other_cards:
            for other_path, other_data in other_cards.items():
                if other_path == str(card_path):
                    continue
                other_ext = other_data.get("qa-extensions", {})
                other_names = {a.get("name") for a in other_ext.get("auditors", [])}
                if name in other_names:
                    raise QAExtensionsCollisionError(
                        f"auditor name {name!r} declarado em 2 cards: "
                        f"{card_path} + {other_path}. Approach A: hard fail "
                        f"sem merge (Decisão 28)."
                    )
```

# engine/qa/scope.py
"""Scope resolution pra forge qa (Phase 0 ingest).

Conversacional (Decisão 10 — sem flags). Resolve via 3-caminhos quando
ambíguo. Reusa pattern de engine.verify (mandamento #3).
"""

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Literal


ScopeType = Literal["feature", "screen", "task", "paranoid"]
_VALID_FEATURE_STATES = {"planning", "implementing", "blocked-on-external", "done"}


class ScopeError(RuntimeError):
    pass


class ScopeAmbiguityError(ScopeError):
    """Mesmo input casa em 2+ scopes — pede 3-caminhos."""


class ScopeMissingError(ScopeError):
    """Scope target não encontrado no projeto."""


@dataclass(frozen=True)
class Scope:
    type: ScopeType
    target: str
    paths: tuple[Path, ...]  # artefatos resolvidos


def resolve_scope(
    raw_target: str,
    *,
    project_root: Path,
    paranoid_max_features: int = 10,
) -> Scope:
    """Resolve scope conversacional. raw_target é o input do user:
    `paranoid` (literal) / feature-slug / TASK-NNNN / screen-id.
    """
    if raw_target == "paranoid":
        features = _list_features_for_paranoid(project_root, cap=paranoid_max_features)
        if not features:
            raise ScopeMissingError(
                "Nenhuma feature em scope detectado pra paranoid. "
                "Rode `forge plan {slug}` antes."
            )
        return Scope(type="paranoid", target="paranoid",
                     paths=tuple(features))

    if raw_target.startswith("TASK-"):
        path = _find_task(raw_target, project_root)
        if path is None:
            raise ScopeMissingError(f"task {raw_target} não encontrada")
        return Scope(type="task", target=raw_target, paths=(path,))

    # feature ou screen? Ambiguity check
    feature_path = _find_feature(raw_target, project_root)
    screen_path = _find_screen(raw_target, project_root)
    if feature_path and screen_path:
        raise ScopeAmbiguityError(
            f"target {raw_target!r} casa em feature E screen. "
            f"3-caminhos: 1) feature {feature_path} / 2) screen {screen_path} / "
            f"3) abortar e re-invocar com slug específico."
        )
    if feature_path:
        return Scope(type="feature", target=raw_target, paths=(feature_path,))
    if screen_path:
        return Scope(type="screen", target=raw_target, paths=(screen_path,))

    raise ScopeMissingError(
        f"target {raw_target!r} não casa em feature, screen, ou task. "
        f"Confira slug ou rode `forge plan` antes."
    )


def _list_features_for_paranoid(root: Path, *, cap: int) -> list[Path]:
    # Pattern espelhado de engine.verify — lista feature dirs com state válido.
    candidates: list[Path] = []
    for state_dir in (root / "docs" / "feature-implementation-workflow" / "features").glob("*"):
        if state_dir.is_dir():
            candidates.append(state_dir)
        if len(candidates) >= cap:
            break
    return candidates


def _find_feature(slug: str, root: Path) -> Path | None:
    p = root / "docs" / "feature-implementation-workflow" / "features" / slug
    return p if p.exists() else None


def _find_screen(screen_id: str, root: Path) -> Path | None:
    for feature_dir in (root / "docs" / "feature-implementation-workflow" / "features").glob("*"):
        candidate = feature_dir / "screens" / f"{screen_id}.yaml"
        if candidate.exists():
            return candidate
    return None


def _find_task(task_id: str, root: Path) -> Path | None:
    for feature_dir in (root / "docs" / "feature-implementation-workflow" / "features").glob("*"):
        candidate = feature_dir / "tasks" / f"{task_id}.yaml"
        if candidate.exists():
            return candidate
    return None

Spec §5.3 + Decisão 30 (sandbox isolation):
  - subprocess.cwd = run_dir/fixtures (não toca projeto real)
  - os.chdir rejeitado via PYTHONSTARTUP guard

import subprocess
import sys
import time
from dataclasses import dataclass, field

from typing import Literal

# PYTHONSTARTUP preload — rejeita os.chdir no subprocess do sandbox.
import os, sys
_original_chdir = os.chdir
def _blocked_chdir(_p, *_a, **_kw):
    raise RuntimeError(
        'os.chdir bloqueado pelo sandbox forge qa (Decisão 30). '
        'CWD foi setado externamente e é imutável.'
    )

"""


class SandboxBreachError(RuntimeError):
    """Validator subprocess tentou escapar do CWD do sandbox."""


@dataclass(frozen=True)
class Fixture:
    name: str
    input_path: Path
    validator_path: Path
    executable: bool = True


SandboxStatus = Literal["ok", "timeout", "skipped-budget", "sandbox-breach", "error"]


@dataclass
class SandboxResult:
    fixture: Fixture
    status: SandboxStatus = "ok"
    exit_code: int | None = None
    stdout: str = ""
    stderr: str = ""
    duration_s: float = 0.0
    error: str = ""


def _validate_paths_inside_sandbox(fixture: Fixture, sandbox_cwd: Path) -> None:
    sandbox_resolved = sandbox_cwd.resolve()
    for p in (fixture.input_path, fixture.validator_path):
        try:
            resolved = p.resolve()
        except OSError:
            raise SandboxBreachError(f"path {p} não resolve")
        if not str(resolved).startswith(str(sandbox_resolved)):
            # validator pode ser absoluto fora (canon), apenas input precisa estar dentro
            if p == fixture.input_path:
                raise SandboxBreachError(
                    f"fixture.input_path {p} fora do sandbox {sandbox_cwd}"
                )


def _hardened_env(startup_path: Path) -> dict[str, str]:
    env = dict(os.environ)
    env["PYTHONSTARTUP"] = str(startup_path)
    env["FORGE_QA_SANDBOX"] = "1"
    return env


def run_sandbox(
    run_dir: Path,
    fixtures: list[Fixture],
    *,
    budget_total_s: float = 60.0,
    per_validator_s: float = 15.0,
) -> list[SandboxResult]:
    """Loop subprocess pra cada fixture. Hardening conforme Decisão 30."""
    sandbox_cwd = run_dir / "fixtures"
    sandbox_cwd.mkdir(parents=True, exist_ok=True)

    # PYTHONSTARTUP guard file
    startup_file = run_dir / "_chdir_guard.py"
    startup_file.write_text(_CHDIR_GUARD, encoding="utf-8")

    results: list[SandboxResult] = []
    started = time.monotonic()

    for fixture in fixtures:
        elapsed = time.monotonic() - started
        if elapsed >= budget_total_s:
            results.append(SandboxResult(fixture=fixture, status="skipped-budget"))
            continue

        try:
            _validate_paths_inside_sandbox(fixture, sandbox_cwd)
        except SandboxBreachError as e:
            results.append(SandboxResult(
                fixture=fixture, status="sandbox-breach", error=str(e),
            ))
            continue

        remaining = min(per_validator_s, budget_total_s - elapsed)
        t0 = time.monotonic()
        try:
            proc = subprocess.run(
                [sys.executable, str(fixture.validator_path), str(fixture.input_path)],
                cwd=sandbox_cwd,
                capture_output=True,
                text=True,
                timeout=remaining,
                env=_hardened_env(startup_file),
            )
            results.append(SandboxResult(
                fixture=fixture,
                status="ok",
                exit_code=proc.returncode,
                stdout=proc.stdout,
                stderr=proc.stderr,
                duration_s=time.monotonic() - t0,
            ))
        except subprocess.TimeoutExpired:
            results.append(SandboxResult(
                fixture=fixture, status="timeout", duration_s=time.monotonic() - t0,
            ))
        except OSError as e:
            results.append(SandboxResult(
                fixture=fixture, status="error", error=str(e),
            ))

    return results
```

# engine/qa/synthesis.py
"""Phase 4 — Synthesis. Dedup + severity assignment + verdict calculation.

Spec §5.4. Reusa Decisão 25 fingerprint canonical-form pra dedup.
"""

from __future__ import annotations

import hashlib
import json
from collections import Counter
from dataclasses import dataclass
from typing import Any, Literal


Severity = Literal["critical", "high", "medium", "low", "info"]
Verdict = Literal["BLOCK", "FLAG", "PASS"]


def canonical_fingerprint(finding: dict[str, Any]) -> str:
    """Decisão 25 — fingerprint estável contra timestamps e cosmetic edits.

    canonical-form = sha256({type, normalized_title, normalized_description,
                              sorted(scope.files)}).
    """
    norm = {
        "type": finding.get("vector", ""),
        "title": (finding.get("title") or "").strip().lower(),
        "description": (finding.get("description") or "").strip().lower(),
        "files": sorted((finding.get("scope") or {}).get("files", [])),
    }
    blob = json.dumps(norm, sort_keys=True, ensure_ascii=False).encode("utf-8")
    return hashlib.sha256(blob).hexdigest()


def dedup_findings(findings: list[dict[str, Any]]) -> list[dict[str, Any]]:
    by_fp: dict[str, dict[str, Any]] = {}
    for f in findings:
        fp = f.get("fingerprint") or canonical_fingerprint(f)
        f["fingerprint"] = fp
        if fp in by_fp:
            # merge evidence — anexa em evidence_extras
            existing = by_fp[fp]
            extras = existing.setdefault("evidence_extras", [])
            extras.append(f.get("evidence", {}))
        else:
            by_fp[fp] = dict(f)
    return list(by_fp.values())


def compute_verdict(findings: list[dict[str, Any]]) -> Verdict:
    """Verdict logic — spec §5.4.

      BLOCK if critical >= 1 or high >= 3
      FLAG  if high in {1,2} or medium >= 3
      PASS  otherwise
    """
    sev_count = Counter(f["severity"] for f in findings)
    if sev_count["critical"] >= 1 or sev_count["high"] >= 3:
        return "BLOCK"
    if sev_count["high"] in (1, 2) or sev_count["medium"] >= 3:
        return "FLAG"
    return "PASS"


@dataclass(frozen=True)
class SynthesisResult:
    findings: list[dict[str, Any]]
    verdict: Verdict
    by_severity: dict[str, int]
    by_vector: dict[str, int]


def synthesize(draft_findings: list[dict[str, Any]]) -> SynthesisResult:
    deduped = dedup_findings(draft_findings)
    verdict = compute_verdict(deduped)
    by_sev = dict(Counter(f["severity"] for f in deduped))
    for k in ("critical", "high", "medium", "low", "info"):
        by_sev.setdefault(k, 0)
    by_vec = dict(Counter(f["vector"] for f in deduped))
    return SynthesisResult(
        findings=deduped, verdict=verdict, by_severity=by_sev, by_vector=by_vec,
    )
```

# engine/qa/emit.py
"""Phase 5 — Emit. Filtra findings actionable, dedup contra rejected,
escreve em proposed-evolutions.yaml.

Spec §5.5.
"""

from __future__ import annotations

import os
from pathlib import Path
from typing import Any

import yaml

_ACTIONABLE_SEVERITIES = {"critical", "high", "medium"}


def load_rejected_fingerprints(project_root: Path) -> set[str]:
    p = project_root / ".claude" / "memory" / "L1" / "proposed-evolutions" / "rejected-fingerprints.yaml"
    if not p.exists():
        return set()
    data = yaml.safe_load(p.read_text(encoding="utf-8")) or {}
    return set(data.get("rejected", []))


def filter_actionable(findings: list[dict[str, Any]]) -> list[dict[str, Any]]:
    return [f for f in findings if f.get("severity") in _ACTIONABLE_SEVERITIES]


def emit_proposed_evolutions(
    findings: list[dict[str, Any]],
    *,
    project_root: Path,
) -> dict[str, Any]:
    """Escreve findings actionable em proposed.yaml. Atomic write. Retorna
    summary dict {written: int, skipped: int, write_failed: bool}.
    """
    rejected = load_rejected_fingerprints(project_root)
    actionable = filter_actionable(findings)

    to_write = [f for f in actionable if f.get("fingerprint") not in rejected]
    skipped = len(actionable) - len(to_write)

    out_dir = project_root / ".claude" / "memory" / "L1" / "proposed-evolutions"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / "proposed.yaml"

    existing = {}
    if out_path.exists():
        existing = yaml.safe_load(out_path.read_text(encoding="utf-8")) or {}

    entries = existing.get("entries", [])
    for f in to_write:
        entries.append({
            "type": f["proposed_evolution"]["type"],
            "fingerprint": f["fingerprint"],
            "summary": f["proposed_evolution"].get("summary", f["title"]),
            "payload": f,
        })

    new_data = {"entries": entries}
    tmp_path = out_path.with_suffix(out_path.suffix + ".tmp")
    write_failed = False
    try:
        tmp_path.write_text(yaml.safe_dump(new_data, sort_keys=False), encoding="utf-8")
        os.replace(tmp_path, out_path)
    except OSError:
        write_failed = True
        if tmp_path.exists():
            tmp_path.unlink(missing_ok=True)

    return {"written": len(to_write), "skipped": skipped, "write_failed": write_failed}
```

# engine/qa/run_id.py
"""Gerador de run_id pra forge qa.

Formato: YYYY-MM-DDTHH-MM-SSZ-<4-char-hex>. UTC garantido. Sufixo
aleatório distingue runs paralelas com mesma timestamp.
"""

from __future__ import annotations

import secrets
from datetime import datetime, timezone


def generate_run_id(*, now: datetime | None = None) -> str:
    ts = now or datetime.now(timezone.utc)
    if ts.tzinfo is None or ts.utcoffset().total_seconds() != 0:
        ts = ts.astimezone(timezone.utc)
    formatted = ts.strftime("%Y-%m-%dT%H-%M-%SZ")
    suffix = secrets.token_hex(2)  # 4 hex chars
    return f"{formatted}-{suffix}"
